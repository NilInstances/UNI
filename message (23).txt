local function Connect(Signal, Callback)
	local Connection = Signal:Connect(Callback)
	Connections[#Connections + 1] = Connection
	return Connection
end

local function AddLoop(Name, Callback)
	if Loops[Name] then
		Loops[Name]:Disconnect()
	end

	Loops[Name] = RunService.Heartbeat:Connect(Callback)
	Connections[#Connections + 1] = Loops[Name]
end

local function RemoveLoop(Name)
	if Loops[Name] then
		Loops[Name]:Disconnect()
		Loops[Name] = nil
	end
end

local function RebuildCmdCache()
	local CacheSize, ReturnsSize = 0, 0

	for _ in Cmds do CacheSize += 1 end
	for _ in Aliases do CacheSize += 1 end

	CmdCache = table.create(CacheSize)
	ReturnsValueCache = {}

	local Seen, Idx, ReturnsIdx = {}, 0, 0

	for Name, Cmd in Cmds do
		Name = strLower(Name)
		if not Seen[Name] then
			Seen[Name] = true
			Idx += 1
			CmdCache[Idx] = Name
			if Cmd.ReturnsValue then
				ReturnsIdx += 1
				ReturnsValueCache[ReturnsIdx] = Name
			end
		end
	end

	for Alias, Target in Aliases do
		Alias = strLower(Alias)
		if not Seen[Alias] then
			Seen[Alias] = true
			Idx += 1
			CmdCache[Idx] = Alias
			local Cmd = Cmds[Target]
			if Cmd and Cmd.ReturnsValue then
				ReturnsIdx += 1
				ReturnsValueCache[ReturnsIdx] = Alias
			end
		end
	end

	table.sort(CmdCache)
	table.sort(ReturnsValueCache)
end

local function GetCmd(Name)
	if not Name then return nil end
	Name = strLower(Name)
	return Cmds[Name] or Cmds[Aliases[Name]]
end

local function GetAutoComplete(Input, OnlyReturnsValue)
	if #Input == 0 then return nil end

	Input = strLower(Input)
	local InputLen = #Input

	for _, CmdName in OnlyReturnsValue and ReturnsValueCache or CmdCache do
		if strSub(CmdName, 1, InputLen) == Input then
			return CmdName
		end
	end

	return nil

end

local function UpdateAutoComplete()
	local Text = CmdBox.Text
	local TextLen = #Text

	if TextLen == 0 then
		AutoCompleteLabel.Text = ""
		CurSuggestion = nil
		CurSuggestionStart = nil
		return
	end

	local SegmentStart, i, InQuotes = 1, 1, false

	while i <= TextLen do
		local Char = strSub(Text, i, i)
		if Char == '"' then
			InQuotes = not InQuotes
		elseif Char == '&' and strSub(Text, i + 1, i + 1) == '&' and not InQuotes then
			SegmentStart = i + 2
			i += 1
		end
		i += 1
	end

	local Segment = strSub(Text, SegmentStart)
	local SegmentLen = #Segment

	local WordCount = 0
	local CurWord = ""
	local CurWordStart = SegmentStart
	local InParens = 0
	InQuotes = false

	for j = 1, SegmentLen do
		local Char = strSub(Segment, j, j)
		local AbsPos = SegmentStart + j - 1

		if Char == '"' and InParens == 0 then
			InQuotes = not InQuotes
			CurWord ..= Char
		elseif Char == '(' and not InQuotes then
			if #CurWord > 0 then
				WordCount += 1
				CurWord = ""
			end
			InParens += 1
		elseif Char == ')' and not InQuotes then
			InParens = math.max(0, InParens - 1)
			if #CurWord > 0 then
				CurWord = ""
			end
		elseif Char == ' ' and not InQuotes and InParens == 0 then
			if #CurWord > 0 then
				WordCount += 1
				CurWord = ""
			end
		elseif Char == ' ' and not InQuotes and InParens > 0 then
			if #CurWord > 0 then
				CurWord = ""
			end
		else
			if #CurWord == 0 then
				CurWordStart = AbsPos
			end
			CurWord ..= Char
		end
	end

	local CurWordLen = #CurWord

	if CurWordLen == 0 then
		AutoCompleteLabel.Text = ""
		CurSuggestion = nil
		CurSuggestionStart = nil
		return
	end

	local OnlyReturnsValue = InParens > 0 or WordCount > 0
	local Suggestion = GetAutoComplete(CurWord, OnlyReturnsValue)

	CurSuggestion = Suggestion
	CurSuggestionStart = CurWordStart

	if Suggestion and Suggestion ~= strLower(CurWord) then
		local Padding = strRep(" ", CurWordStart - 1 + CurWordLen)
		AutoCompleteLabel.Text = Padding .. strSub(Suggestion, CurWordLen + 1)
	else
		AutoCompleteLabel.Text = ""
		CurSuggestion = nil
	end
end

local function AcceptAutoComplete()
	if not CurSuggestion or not CurSuggestionStart then return false end

	local Text = CmdBox.Text
	local BeforeWord = strSub(Text, 1, CurSuggestionStart - 1)

	CmdBox.Text = BeforeWord .. CurSuggestion .. " "

	task.delay(0, function()
		CmdBox.Text = strGsub(CmdBox.Text, "\t", "")
	end)

	CmdBox.CursorPosition = #CmdBox.Text + 1
	CurSuggestion = nil
	CurSuggestionStart = nil
	AutoCompleteLabel.Text = ""

	return true

end

local function ToggleCmdBox(ForceState)
	local NewState = ForceState ~= nil and ForceState or not CmdBox.Visible
	CmdBox.Visible = NewState

	if NewState then
		HistoryIdx = 0
		CurSuggestion = nil
		CurSuggestionStart = nil
		AutoCompleteLabel.Text = ""
		CmdBox:CaptureFocus()

		CmdBox:GetPropertyChangedSignal("Text"):Once(function()
			CmdBox.Text = ""
		end)
	else
		CmdBox:ReleaseFocus()
	end

end

local function GetRawStr(Input)
	local Replacements = {
		["&amp;"] = "&",
		["&lt;"] = "<",
		["&gt;"] = ">",
		["&quot;"] = '"',
		["&apos;"] = "'",
	}


	for Ent, Char in Replacements do
		Input = strGsub(Input, Ent, Char)
	end

	return Input

end

local function ParseArgs(Input)
	local Args = {}
	local ArgsIdx = 0
	local Cur = ""
	local InQuotes = false
	local InParens = 0
	local CurQuoted = false
	local CurIsParen = false
	local InputLen = #Input

	for i = 1, InputLen do
		local Char = strSub(Input, i, i)

		if Char == '"' and InParens == 0 then
			if not InQuotes then
				CurQuoted = true
			end
			InQuotes = not InQuotes
		elseif Char == '(' and not InQuotes then
			if InParens == 0 then
				if #Cur > 0 then
					ArgsIdx += 1
					Args[ArgsIdx] = {Cur, CurQuoted, false}
					Cur = ""
					CurQuoted = false
				end
				CurIsParen = true
			else
				Cur ..= Char
			end
			InParens += 1
		elseif Char == ')' and not InQuotes then
			InParens -= 1
			if InParens == 0 then
				if #Cur > 0 then
					ArgsIdx += 1
					Args[ArgsIdx] = {Cur, false, CurIsParen}
					Cur = ""
					CurIsParen = false
				end
			else
				Cur ..= Char
			end
		elseif Char == ' ' and not InQuotes and InParens == 0 then
			if #Cur > 0 then
				ArgsIdx += 1
				Args[ArgsIdx] = {Cur, CurQuoted, CurIsParen}
				Cur = ""
				CurQuoted = false
				CurIsParen = false
			end
		else
			Cur ..= Char
		end
	end

	if #Cur > 0 then
		ArgsIdx += 1
		Args[ArgsIdx] = {Cur, CurQuoted, CurIsParen}
	end

	return Args
end



local function SplitCmds(Input)
	local Result = {}
	local ResultIdx = 0
	local Cur = ""
	local InQuotes = false
	local i = 1
	local InputLen = #Input

	while i <= InputLen do
		local Char = strSub(Input, i, i)

		if Char == '"' then
			InQuotes = not InQuotes
			Cur ..= Char
		elseif Char == '&' and strSub(Input, i + 1, i + 1) == '&' and not InQuotes then
			local Trimmed = strMatch(Cur, "^%s*(.-)%s*$")
			if #Trimmed > 0 then
				ResultIdx += 1
				Result[ResultIdx] = Trimmed
			end
			Cur = ""
			i += 1
		else
			Cur ..= Char
		end

		i += 1
	end

	local Trimmed = strMatch(Cur, "^%s*(.-)%s*$")
	if #Trimmed > 0 then
		ResultIdx += 1
		Result[ResultIdx] = Trimmed
	end

	return Result

end

local function SubVars(Args)
	local Result = table.create(#Args)
	for i, Arg in Args do
		local Value = Arg[1]

		if type(Value) == "string" and strSub(Value, 1, 1) == "$" then
			Result[i] = {Vars[strSub(Value, 2)] or Value, Arg[2], Arg[3]}
		else
			Result[i] = Arg
		end
	end
	return Result
end

local ExecCmd

local function EvalParenExprs(Args)
	local Result = table.create(#Args)
	for i, Arg in Args do
		if Arg[3] and type(Arg[1]) == "string" then
			local ReturnValue = ExecCmd(Arg[1])
			Result[i] = {ReturnValue, true}
		else
			Result[i] = {Arg[1], Arg[2]}
		end
	end
	return Result
end

local function ProcessArgs(Args, StartIdx)
	StartIdx = StartIdx or 1
	local Result = {}
	local ResultIdx = 0
	local i = StartIdx
	local ArgsLen = #Args

	while i <= ArgsLen do
		local Arg = Args[i]
		local ArgValue = Arg[1]

		if not Arg[2] and type(ArgValue) == "string" then
			local Cmd = GetCmd(ArgValue)

			if Cmd and Cmd.ReturnsValue then
				local SubResult = ProcessArgs(Args, i + 1)
				local ReturnValue = Cmd.Callback(table.unpack(SubResult))
				ResultIdx += 1
				Result[ResultIdx] = ReturnValue
				return Result
			end
		end

		ResultIdx += 1
		Result[ResultIdx] = ArgValue
		i += 1
	end

	return Result

end

ExecCmd = function(CmdStr)
	local Name, ArgsStr = strMatch(CmdStr, "^(%S+)%s*(.*)")
	if not Name then return end

	local Cmd = GetCmd(Name)

	if Cmd then
		return Cmd.Callback(table.unpack(ProcessArgs(EvalParenExprs(SubVars(ParseArgs(ArgsStr or ""))))))
	end

	return nil
end

local function ExecCmdChain(FullCmd)
	if #FullCmd == 0 then return end

	CmdHistory[#CmdHistory + 1] = FullCmd

	if #CmdHistory > MaxHistory then
		table.remove(CmdHistory, 1)
	end

	task.spawn(function()
		for _, CmdStr in SplitCmds(FullCmd) do
			ExecCmd(CmdStr)
		end
	end)

end

local function GetPlayers(Arg)
	if not Arg then return {} end

	local Selector = type(Arg) == "table" and Arg[1] or tostring(Arg)
	local Quoted = type(Arg) == "table" and Arg[2] or false
	local LowerSelector = strLower(Selector)
	local LocalCharacter = LocalPlayer.Character
	local LocalRoot = LocalCharacter and LocalCharacter:FindFirstChild("HumanoidRootPart")

	if Quoted then
		local Result = {}
		for _, Player in Players:GetPlayers() do
			if strFind(strLower(Player.Name), LowerSelector, 1, true) or 
				strFind(strLower(Player.DisplayName), LowerSelector, 1, true) then
				Result[#Result + 1] = Player
			end
		end
		return Result
	end

	local Selectors = {
		me = function()
			return {LocalPlayer}
		end,

		all = function()
			return Players:GetPlayers()
		end,

		others = function()
			local Result = {}
			for _, Player in Players:GetPlayers() do
				if Player ~= LocalPlayer then
					Result[#Result + 1] = Player
				end
			end
			return Result
		end,

		random = function()
			local All = Players:GetPlayers()
			return #All > 0 and {All[math.random(#All)]} or {}
		end,

		nearest = function()
			if not LocalRoot then return {} end
			local Nearest, NearestDist = nil, math.huge
			for _, Player in Players:GetPlayers() do
				if Player ~= LocalPlayer then
					local Root = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
					if Root then
						local Dist = (Root.Position - LocalRoot.Position).Magnitude
						if Dist < NearestDist then
							Nearest, NearestDist = Player, Dist
						end
					end
				end
			end
			return Nearest and {Nearest} or {}
		end,

		farthest = function()
			if not LocalRoot then return {} end
			local Farthest, FarthestDist = nil, 0
			for _, Player in Players:GetPlayers() do
				if Player ~= LocalPlayer then
					local Root = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
					if Root then
						local Dist = (Root.Position - LocalRoot.Position).Magnitude
						if Dist > FarthestDist then
							Farthest, FarthestDist = Player, Dist
						end
					end
				end
			end
			return Farthest and {Farthest} or {}
		end,
	}

	Selectors.closest = Selectors.nearest

	if Selectors[LowerSelector] then
		return Selectors[LowerSelector]()
	end

	local Result = {}
	for _, Player in Players:GetPlayers() do
		if strFind(strLower(Player.Name), LowerSelector, 1, true) or 
			strFind(strLower(Player.DisplayName), LowerSelector, 1, true) then
			Result[#Result + 1] = Player
		end
	end
	return Result

end

Cmds.save = {
	Aliases = {"savedata", "writedata"},
	Callback = function(Name, NoNotify)
		Name = (Name and Name ~= "") and Name or "Config"

		local Data = {
			Keybinds = Keybinds,
			Aliases = Aliases,
			Vars = Vars,
			Prefixes = Prefixes,
			ToggleKey = ToggleKey.Name,
		}

		local Success, Encoded = pcall(HttpService.JSONEncode, HttpService, Data)

		if Success then
			writefile(`AdvanCMD/Configs/{Name}.json`, Encoded)

			if not NoNotify then
				Notify({Title = "Config", Text = "Successfully saved config!"})
			end
		else
			Notify({Title = "Config", Text = "An error occurred!"})
		end
	end,

}

Cmds.load = {
	Aliases = {"loaddata"},
	Callback = function(Name, NoNotify)
		Name = (Name and Name ~= "") and Name or "Config"

		if not isfile(`AdvanCMD/Configs/{Name}.json`) then
			return Notify({Title = "Config", Text = "Config doesn't exist!"})
		end

		local Success, Decoded = pcall(HttpService.JSONDecode, HttpService, readfile(`AdvanCMD/Configs/{Name}.json`))

		if Success then
			Keybinds = Decoded.Keybinds or {}
			Aliases = Decoded.Aliases or {}
			Vars = Decoded.Vars or {}
			Prefixes = Decoded.Prefixes or {";"}
			if Decoded.ToggleKey then
				ToggleKey = Enum.KeyCode[Decoded.ToggleKey] or ToggleKey
			end
			RebuildCmdCache()

			if not NoNotify then
				Notify({Title = "Config", Text = "Successfully loaded config!"})
			end
		else
			Notify({Title = "Config", Text = "An error occurred!"})
		end
	end,

}

	Aliases = {"sh", "hop"},
	Callback = function()
		local Servers = HttpService:JSONDecode(game:HttpGet(`https://games.roblox.com/v1/games/{game.PlaceId}/servers/Public?sortOrder=Asc&limit=100`))
		for _, Server in Servers.data or {} do

				TeleportService:TeleportToPlaceInstance(game.PlaceId, Server.id, LocalPlayer)
				return
			end
		end
		Notify({Text = "No available servers found!"})
	end

	Callback = function(Offset)
		local Idx = #CmdHistory - (tonumber(Offset) or 1)
		if CmdHistory[Idx] then
			ExecCmdChain(CmdHistory[Idx])
		end
	end,
}

Cmds.history = {
	Callback = function(Count)
		Count = tonumber(Count) or 10
		local Start = math.max(1, #CmdHistory - Count + 1)
		for i = Start, #CmdHistory do
			print(i .. ": " .. CmdHistory[i])
		end
	end,
}

Cmds.math = {
	Aliases = {"calc", "calculate"},
	ReturnsValue = true,
	Callback = function(...)
		local Expr = table.concat({...}, " ")
		local Fn, Err = loadstring("return " .. Expr)
		if Fn then
			local Success, Result = pcall(Fn)
			if Success then
				print(`Result: {Result}`)
				return Result
			end
		end
		return Err
	end,
}

Connect(InputService.InputBegan, function(Input, Processed)
	if Processed then return end

	if Input.KeyCode == ToggleKey then
		ToggleCmdBox()
		return
	end

	local KeyName = strLower(Input.KeyCode.Name)
	if Keybinds[KeyName] then
		for CmdName, Args in Keybinds[KeyName] do
			local CmdData = GetCmd(CmdName)
			if CmdData then
				task.spawn(CmdData.Callback, table.unpack(Args))
			end
		end
	end

end)

Connect(LocalPlayer.Chatted, function(Message)
	Message = GetRawStr(Message)

	for _, Prefix in Prefixes do
		if strSub(Message, 1, #Prefix) == Prefix then
			ExecCmdChain(strSub(Message, #Prefix + 1))
			return
		end
	end

end)

CmdBox.FocusLost:Connect(function(EnterPressed)
	if EnterPressed and #CmdBox.Text > 0 then
		ExecCmdChain(CmdBox.Text)
		CmdBox.Text = ""
	end
	CmdBox.Visible = false
	AutoCompleteLabel.Text = ""
	CurSuggestion = nil
	CurSuggestionStart = nil
end)

CmdBox:GetPropertyChangedSignal("Text"):Connect(function()
	HistoryIdx = 0
	UpdateAutoComplete()
end)

InputService.InputBegan:Connect(function(Input)
	if not CmdBox:IsFocused() then return end

	if Input.KeyCode == Enum.KeyCode.Tab then
		AcceptAutoComplete()
	elseif Input.KeyCode == Enum.KeyCode.Right then
		if CurSuggestion and CmdBox.CursorPosition > #CmdBox.Text then
			AcceptAutoComplete()
		end
	elseif Input.KeyCode == Enum.KeyCode.Up then
		if #CmdHistory > 0 then
			HistoryIdx = math.min(HistoryIdx + 1, #CmdHistory)
			CmdBox.Text = CmdHistory[#CmdHistory - HistoryIdx + 1] or ""
			CmdBox.CursorPosition = #CmdBox.Text + 1
		end
	elseif Input.KeyCode == Enum.KeyCode.Down then
		if HistoryIdx > 0 then
			HistoryIdx -= 1
			CmdBox.Text = HistoryIdx > 0 and CmdHistory[#CmdHistory - HistoryIdx + 1] or ""
			CmdBox.CursorPosition = #CmdBox.Text + 1
		end
	elseif Input.KeyCode == Enum.KeyCode.Escape then
		CmdBox.Text = ""
		CmdBox:ReleaseFocus()
	end

end)

local function Unload()
	pcall(function()
		if isfile("AdvanCMD/Configs/Autoload") then
			Cmds.save.Callback(readfile("AdvanCMD/Configs/Autoload"), true)
		end
	end)

	for State, Value in ToggleStates do
		if Value then
			ToggleStates[State] = true
			local UnCmd = Cmds["un" .. strLower(State)]
			if UnCmd then
				pcall(UnCmd.Callback)
			end
		end
	end

	for _, Connection in Connections do
		if Connection then
			Connection:Disconnect()
		end
	end

	for Name in Loops do
		RemoveLoop(Name)
	end

	if ScreenGui then
		ScreenGui:Destroy()
	end

	getgenv().AdvanCMD = nil

end

getgenv().AdvanCMD = {
	Connections = Connections,
	Loops = Loops,
	ToggleStates = ToggleStates,
	Cmds = Cmds,
	Aliases = Aliases,
	Keybinds = Keybinds,
	Vars = Vars,
	ExecCmd = ExecCmd,
	ExecCmdChain = ExecCmdChain,
	GetPlayers = GetPlayers,
	Notify = Notify,
	Unload = Unload,
}
